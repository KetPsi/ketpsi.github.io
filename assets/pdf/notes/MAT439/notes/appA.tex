\section{Tensor Products}
\begin{definition}
    Let \(\mathcal{V}, \mathcal{W}\) be vector spaces over a field \(\mathbb{K}\). The \emph{tensor product} of \(\mathcal{V}\) and \(\mathcal{W}\), is a pair \((V \otimes W, \otimes)\), where 
    \begin{itemize}
        \item \(\mathcal{V} \otimes \mathcal{W}\) is a vector space over \(\mathbb{K}\),
        \item \(\otimes: \mathcal{V} \otimes \mathcal{W} \longrightarrow \mathcal{V} \otimes \mathcal{W}\) is bilinear,
    \end{itemize}
    such that for every vector space \(\mathcal{U}\) and every bilinear map \(f:\mathcal{V} \times \mathcal{W} \longrightarrow \mathcal{U}\), there exists a unique linear map \(\tilde{f}: \mathcal{V} \otimes \mathcal{W} \longrightarrow U\) with \(f = \tilde{f} \circ \otimes\).
center\[\begin{tikzcd}
    {\mathcal{V} \times \mathcal{W}} && {\mathcal{V} \otimes \mathcal{W}} \\
	\\
                                     && \mathcal{U}
	\arrow["\otimes", from=1-1, to=1-3]
	\arrow["f"', from=1-1, to=3-3]
	\arrow["{\exists!\tilde{f}}", dashed, from=1-3, to=3-3]
\end{tikzcd}\]
\end{definition}
\begin{theorem}
    A tensor product of \(\mathcal{V}\) and \(mathcal{W}\) exists.
\end{theorem}
\begin{proof}
    Let \(F(\mathcal{V} \times \mathcal{W})\) be the free vector space over the set \(\mathcal{V} \times \mathcal{W}\). Elements of \(F(\mathcal{V} \times \mathcal{W})\) are finite formal sums \[\sum_i a_i(v_i, w_i), \quad a_i \in \mathbb{K}.\]
    Let \(R \subset F(\mathcal{V} \times \mathcal{W})\) be the subspace generated by all elements of the form:
    \[(v + v', w) - (v, w) - (v', w),\]
    \[(v, w + w') - (v, w) - (v, w'),\]
    \[(\lambda v, w) - \lambda(v, w),\]
    \[(v, \lambda w) - \lambda(v, w),\] for all \(v, v' \in \mathcal{V}, w, w' \in \mathcal{W}, \lambda \in \mathbb{K}\).
    
    Define \(\mathcal{V} \otimes \mathcal{W} := F/R.\). 

    Let \(\otimes: \mathcal{V} \times \mathcal{W} \longrightarrow \mathcal{V} \otimes \mathcal{W}, \quad (v, w) \mapsto (v,w) + R\). This map is bilinear by construciton.
\end{proof}
